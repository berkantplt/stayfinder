Genel Tanitim
GENEL TANITIM
Acapulco Resort & Convention & SPA, Kuzey Kıbrıs otelleri içerisinde, tatil keyfini dünya standartlarında ve servis kalitesiyle Kıbrıs'a getiren, Akdeniz kültürü ile, gelen konuklarına gerek iş, gerek tatil amacı ile beş yıldızlı ve dostça bir servis sağlamaktadır. Otel hizmetleri, bir kilometre uzunluğundaki özel sahili, yetişkin ve çocuk havuzu, deniz manzaralı ve klimalı otel odaları, sahil barı, restoranları, spor salonu, tenis, plaj voleybolu, golf, ısıtmalı havuz, sıcak banyo, SPA, toplantı ve konferans odaları, hızlı erişimli internet, casino ve hediyelik eşya dükkanlarını içermektedir.
Single ,Double ve Triple Bay Rezervasyonları kabul edilmemektedir
Birbirinden konforlu odaları mevcut olup Villa oda 110, PineCourt/1 156, Pine Court/2 248, Ana Bina 255 ( Toplam 659 Adet Balkonlu Oda), Garden Junior 32, Garden Suite 11, Lagoon Suite 14 adet olmak üzere toplam 826 odası bulunmaktadır.

Oda Sayısı

Genel Uyarılar
Tesis odalarına giriş saati 15:00 ile başlar, çıkış günü ise tesisten ayrılış saatine bakılmaksızın odaların en geç saat 11:00 'de boşaltılması gerekmekte olup tesisten alınan ekstra yiyecek- içecek ve sistem dışı hizmet ücretleri konuklara aittir.
Taze sıkılmış meyve suları, Türk kahvesi ve import içecekler ücretlidir.
Otelde hizmet veren açık alanların kullanımı mevsim koşullarına bağlıdır.
Aile odaları veya bazı özel odalar haricindeki odalarda yapılacak 2+2 veya 3+1 konaklamalar, odada sıkışıklık yaratabilir ve verilen ilave yataklar sabit yatak konforunda olmayabilir.
*1 Haziran 2024 Tarihinde hizmete girecek olan Lagoon Suite odalarda 17 yaş ve üstü misafirlerin konaklaması kabul edilecektir.

let isSyncing = false;
function syncRange(fromPicker, toPicker) {
if (isSyncing) return;
const start = fromPicker.getStartDate();
const end = fromPicker.getEndDate();
if (!start || !end) return;
const tStart = toPicker.getStartDate?.();
const tEnd = toPicker.getEndDate?.();
if (tStart && tEnd && tStart.isSame(start, 'day') && tEnd.isSame(end, 'day')) return;
isSyncing = true;
toPicker.setDateRange(start.toJSDate(), end.toJSDate());
isSyncing = false;
}

async function loadSimilarHotels() {
const swiperEl = document.getElementById("similarHotelsSwiper");
const wrapper = document.getElementById("similarHotelsWrapper");
const hotelId = swiperEl?.dataset?.hotelId;
if (!swiperEl || !wrapper || !hotelId) return;
// Loading (basit)
wrapper.innerHTML = `Yükleniyor...
`;
try {
const res = await fetch(`/Hotels/SearchHotel/GetSimilar?Region=${hotelId}`, {
headers: { "Accept": "application/json" }
});
if (!res.ok) throw new Error("API error: " + res.status);
const json = await res.json();
const list = json.results ?? [];
if (!list.length) {
wrapper.innerHTML = `Benzer otel bulunamadı.
`;
return;
}
wrapper.innerHTML = list.map(similarHotelSlide).join("");
// Swiper güncelle / init
// Eğer swiper zaten init edildiyse:
if (swiperEl.swiper) {
swiperEl.swiper.update();
} else {
// Swiper init (projendeki mevcut ayarlarına göre düzenle)
new Swiper("#similarHotelsSwiper", {
slidesPerView: 4,
spaceBetween: 16,
navigation: {
nextEl: "#similarHotelsSwiper .swiper-button-next",
prevEl: "#similarHotelsSwiper .swiper-button-prev",
},
breakpoints: {
0: { slidesPerView: 1.2 },
576: { slidesPerView: 2.2 },
992: { slidesPerView: 4 }
}
});
}
} catch (err) {
console.error(err);
wrapper.innerHTML = `Benzer oteller yüklenemedi.
`;
}
}

let reviewPageState = {
currentPage: 1,
hotelId: "57",
isLoading: false
};
async function loadHotelReviews(isLoadMore = false) {
const commentListContainer = document.querySelector(".comment-list");
const loadMoreContainer = document.querySelector(".load-more-container");
const loadMoreBtn = document.getElementById("btn-load-more");
if (!commentListContainer || !reviewPageState.hotelId || reviewPageState.hotelId == 0) return;
if (reviewPageState.isLoading) return;
reviewPageState.isLoading = false;
// İlk yüklemede loading animasyonu basıyoruz
if (!isLoadMore) {
commentListContainer.innerHTML = `Yorumlar yükleniyor...
`;
loadMoreContainer.style.display = "none";
} else {
loadMoreBtn.disabled = true;
loadMoreBtn.innerText = "Yükleniyor...";
}
try {
const res = await fetch(`/Hotels/Reviews/${reviewPageState.hotelId}?page=${reviewPageState.currentPage}`, {
headers: { "Accept": "application/json" }
});
if (!res.ok) throw new Error("Yorumlar yüklenirken hata oluştu: " + res.status);
const json = await res.json();
const list = json.reviews ?? [];
// İlk sayfa yüklenirken iskele/loading elementini temizle
if (!isLoadMore) {
const skeleton = document.getElementById("review-skeleton");
if (skeleton) skeleton.remove();
}
if (!list.length && !isLoadMore) {
commentListContainer.innerHTML = `Bu otel için henüz yorum yapılmamış.
`;
return;
}
// Gelen verileri haritalandırıp HTML üretiyoruz
const htmlContent = list.map(review => {
let ratingText = 'Başarılı';
if (review.Rating >= 5) ratingText = 'Mükemmel';
else if (review.Rating >= 4) ratingText = 'Çok İyi';
else if (review.Rating >= 3) ratingText = 'İyi';
return `
${review.FullNameMasked}
${review.FormattedDate}
${Number(review.Rating).toFixed(1)} ${ratingText}
${review.Title ? `${review.Title}
` : ''}
${review.Comment}
Genel Puan ${Number(review.Rating).toFixed(1)}
`;
}).join("");
// Eğer "Daha Fazla" denmişse üzerine ekle, yoksa sıfırdan bas
if (isLoadMore) {
commentListContainer.insertAdjacentHTML('beforeend', htmlContent);
} else {
commentListContainer.innerHTML = htmlContent;
}
// Sayfalamanın devamı var mı kontrolü
if (json.has_more) {
loadMoreContainer.style.display = "block";
loadMoreBtn.disabled = false;
loadMoreBtn.innerText = "Daha Fazla Yorum Göster";
} else {
loadMoreContainer.style.display = "none";
}
} catch (err) {
console.error(err);
if (!isLoadMore) {
commentListContainer.innerHTML = `Yorumlar yüklenirken bir hata oluştu.
`;
} else {
alert("Yeni yorumlar yüklenirken bir sorun oluştu.");
loadMoreBtn.disabled = false;
loadMoreBtn.innerText = "Daha Fazla Yorum Göster";
}
} finally {
reviewPageState.isLoading = false;
}
}
// Tetikleyicileri (Event Listeners) bağla
document.addEventListener("DOMContentLoaded", () => {
// İlk yükleme
loadHotelReviews(false);
// Daha fazla yükle butonu tetikleyicisi
const loadMoreBtn = document.getElementById("btn-load-more");
if (loadMoreBtn) {
loadMoreBtn.addEventListener("click", () => {
reviewPageState.currentPage += 1;
loadHotelReviews(true);
});
}
});
document.getElementById('addCommentForm')?.addEventListener('submit', async function (e) {
e.preventDefault();
const form = this;
// HTML5 Doğrulama kontrolü
if (!form.checkValidity()) {
form.classList.add('was-validated');
return;
}
// Mevcut global loading modalını açıyoruz
// if (typeof showLoading === 'function') showLoading();
// Form verilerini topluyoruz
const formData = new FormData(form);
try {
// İleride yazacağın veya tanımladığın yorum kaydetme route adresi (Örn: /Hotels/SaveReview)
const res = await fetch('/Hotels/SaveReview', {
method: 'POST',
body: formData,
headers: { 'X-Requested-With': 'XMLHttpRequest' }
});
if (!res.ok) throw new Error("Sunucu hatası: " + res.status);
const json = await res.json();
if (json.status) {
// Modalı kapat
const modalEl = document.getElementById('addCommentModal');
const modalInstance = bootstrap.Modal.getInstance(modalEl);
if (modalInstance) modalInstance.hide();
// Formu sıfırla
form.reset();
form.classList.remove('was-validated');
// Başarılı uyarısı (İsteğe bağlı alert yerine projendeki toast yapısını koyabilirsin)
alert(json.message ?? 'Yorumunuz başarıyla alındı. Onaylandıktan sonra listelenecektir.');
// Yorum listesini dinamik olarak yeniden güncelle
if (typeof loadHotelReviews === 'function') {
loadHotelReviews();
}
} else {
alert(json.message ?? 'Yorum kaydedilirken bir hata oluştu.');
}
} catch (err) {
console.error(err);
alert('Sistemsel bir hata oluştu, lütfen daha sonra tekrar deneyiniz.');
} finally {
// Global loading modalını kapatıyoruz
if (typeof hideLoading === 'function') hideLoading();
}
});

document.addEventListener("DOMContentLoaded", function () {
let lazyImages = document.querySelectorAll("img.lazy-load");
let observer = new IntersectionObserver((entries, observer) => {
entries.forEach(entry => {
if (entry.isIntersecting) {
let img = entry.target;
img.src = img.getAttribute("data-src");
img.classList.remove("lazy-load");
observer.unobserve(img);
}
});
});
lazyImages.forEach(img => observer.observe(img));
});
